    def get_price(self, symbol: str) -> float:
        """Get current price from bridge or MT4"""
        try:
            # Try price bridge first
            price = self.price_bridge.get_price(symbol)
            if price and price > 0:
                return price
            
            # Try MT4
            if self.mt4:
                result = self.mt4._send({"command": "PRICE", "symbol": symbol})
                if result and isinstance(result, dict):
                    bid = result.get('bid', 0)
                    ask = result.get('ask', 0)
                    if bid > 0 and ask > 0:
                        return (bid + ask) / 2
            
            # Fallback
            return 0
        except Exception as e:
            logger.debug(f"Price error for {symbol}: {e}")
            return 0
    def get_all_mt4_prices(self) -> Dict:
        """
        Get ALL prices from MT4 in ONE request (like dashboard does).
        This ensures prices are consistent across the system.
        """
        try:
                if not self.mt4:
                        return {}
                
                # Send ACCOUNT command (same as dashboard)
                result = self.mt4._send({"command": "ACCOUNT"})
                
                if not result or not isinstance(result, dict):
                        return {}
                
                # Map MT4 keys to symbol names
                price_map = {
                        'EURUSD': result.get('EURUSD', 0),
                        'GBPUSD': result.get('GBPUSD', 0),
                        'USDJPY': result.get('USDJPY', 0),
                        'GOLD': result.get('GOLD', 0),
                        'SILVER': result.get('SILVER', 0),
                        '#NASDAQ100': result.get('NAS100', 0),
                        '#DJ30': result.get('DJ30', 0),
                        '#S&P500': result.get('SP500', 0),
                        'BRENT_OIL': result.get('BRENT', 0),
                        'CrudeOIL': result.get('CRUDE', 0),
                }
                
                # Also check for direct keys
                for symbol in ['EURUSD', 'GBPUSD', 'USDJPY', 'GOLD', 'SILVER', 
                                          '#NASDAQ100', '#DJ30', '#S&P500', 'BRENT_OIL', 'CrudeOIL']:
                        if symbol in result and symbol not in price_map:
                                price_map[symbol] = result.get(symbol, 0)
                
                return price_map
                
        except Exception as e:
                logger.warning(f"Error getting all MT4 prices: {e}")
                return {}

    def _get_price_data(self, symbol: str) -> Dict:
        """Get price data using price_bridge (same as dashboard)."""
        
        # Use price_bridge
        from price_bridge import price_bridge
        
        price = price_bridge.get_price(symbol)
        bid = price_bridge.get_bid(symbol)
        ask = price_bridge.get_ask(symbol)
        
        if price > 0:
                return {
                        'price': price,
                        'bid': bid if bid > 0 else price,
                        'ask': ask if ask > 0 else price,
                        'spread': (ask - bid) if bid > 0 and ask > 0 else 0.0002,
                        'source': 'BRIDGE'
                }
        
        return {
                'price': 0,
                'bid': 0,
                'ask': 0,
                'spread': 0,
                'source': 'NONE'
        }

    def build_market_data(self) -> Dict:
        """Build complete market data for analysis"""
        market_data = {}
        
        # Get prices from price_bridge
        prices = self.get_all_prices()
        
        # Add prices to market_data
        for pair in self.pairs:
                if pair in prices and prices[pair] > 0:
                        market_data[pair] = prices[pair]
                        market_data[f'{pair}_price'] = prices[pair]
        
        # Add current price
        if 'EURUSD' in market_data and market_data['EURUSD'] > 0:
                market_data['current_price'] = market_data['EURUSD']
                market_data['price'] = market_data['EURUSD']
                market_data['pair'] = 'EURUSD'
        
        # Add engine state
        engine_state = self.engine.get_status() if self.engine else {}
        market_data['engine_state'] = engine_state
        
        # Add engine predictions for each pair
        for pair in self.pairs:
                if pair in self.brokers:
                        gear_pred = self.engine.get_gear_prediction(
                                pair, 
                                self.brokers[pair].gear_ratio
                        )
                        market_data[f'{pair}_gear_prediction'] = gear_pred
        
        # Add timestamp
        market_data['timestamp'] = datetime.now().isoformat()
        
        if prices:
                logger.info(f"📊 Built market data with {len(prices)} prices")
        
        return market_data
    def process_market_data(self, market_data: Dict) -> Dict:
        """
        Process market data through all components
        """
        self.cycle_count += 1
        if market_data is None:
            market_data = self.build_market_data()
        results = {
            'cycle': self.cycle_count,
            'timestamp': datetime.now().isoformat(),
            'engine': {},
            'anomalies': {},
            'agents': {},
            'signals': {},
            'consensus': {},
            'trade': {},
            'positions': {}
        }
        
        try:
            # 1. Update engine
            if self.engine:
                results['engine'] = self.engine.update_engine_state(market_data)
                logger.debug(f"Engine: {results['engine'].get('engine_direction')} @ {results['engine'].get('engine_speed', 0):.3f}")
                
            
            # 2. Check for anomalies
            if self.anomaly_detector:
                results['anomalies'] = self.anomaly_detector.update(market_data)
                # If protection active, skip trading
                if results['anomalies'].get('protection_active', False):
                    results['consensus'] = {'signal': 'HOLD', 'reason': 'Protection active'}
                    results['trade'] = {'status': 'SKIPPED', 'reason': 'Protection active'}
                    self.last_results = results
                    return results
            
            # 3. Process agents
            agent_results = {}
            
            for agent in self.agents:
                try:
                    signal_data = {
                        'pair': market_data.get('pair', 'EURUSD'),
                        'price': market_data.get('EURUSD', 0),
                        'current_price': market_data.get('EURUSD', 0),
                        'symbol': market_data.get('symbol', 'EURUSD'),
                    }
                    for pair in self.pairs:
                        if pair in market_data:
                            signal_data[pair] = market_data[pair]
                    
                    # Add candles if available
                    if 'candles' in market_data:
                        signal_data['candles'] = market_data['candles']
                    
                    result = agent.analyze(signal_data)
                    agent_results[agent.name] = result
                    
                    logger.debug(f"{agent.name}: {result.get('vote', 'HOLD')} ({result.get('confidence', 50)}%)")
                    
                except Exception as e:
                    logger.error(f"Agent {agent.name} error: {e}")
                    agent_results[agent.name] = {
                        'vote': 'HOLD', 
                        'confidence': 50, 
                        'reasoning': f'Error: {e}'
                    }
            results['agents'] = agent_results
            
            # 4. Generate signals
            signals = self._generate_signals(agent_results, market_data)
            results['signals'] = signals
            
            # 5. Generate consensus
            results['consensus'] = self._generate_consensus(signals)
            
            # 6. Execute trade if consensus is strong
            consensus = results['consensus']
            min_confidence = self.config.get('min_confidence', 60)
            if consensus['signal'] != 'HOLD' and consensus['confidence'] >= min_confidence:
                pair = market_data.get('pair', 'EURUSD')
                price = market_data.get(pair, 0)
                if price > 0:
                    # Run Monte Carlo simulation
                    mc_results = self.monte_carlo.simulate_entry(
                        current_price=price,
                        direction=consensus['signal']
                    )
                    
                    if mc_results and mc_results.get('entry_confidence', 0) > 50:
                        # Execute trade
                        results['trade'] = self.execution_engine.execute_signal(
                            consensus,
                            market_data,
                            position_size=1.0
                        )
                        logger.info(f"💼 Trade executed: {pair} {consensus['signal']} @ {price:.4f}")
                    else:
                        results['trade'] = {
                            'status': 'SKIPPED', 
                            'reason': f'Monte Carlo confidence: {mc_results.get("entry_confidence", 0):.1f}%'
                        }
                else:
                    results['trade'] = {'status': 'SKIPPED', 'reason': 'No price available'}
            else:
                results['trade'] = {
                    'status': 'SKIPPED', 
                    'reason': f'Low confidence: {consensus["confidence"]:.1f}% < {min_confidence}%' if consensus['signal'] != 'HOLD' else 'HOLD signal'
                }
            
            # 7. Update P&L
            if self.execution_engine:
                results['pnl'] = self.execution_engine.update_pnl(market_data)
            
            # self.last_cycle_time = datetime.now()
            self._log_decision(results)
            
        except Exception as e:
            logger.error(f"Error processing cycle: {e}")
            import traceback
            traceback.print_exc()
            results['error'] = str(e)
        
        self.last_results = results
        return results
    
    def _generate_signals(self, agent_results: Dict, market_data: Dict) -> Dict:
        """
        Generate signals from agent results
        """
        signals = {}
        
        for agent_name, result in agent_results.items():
            pair = result.get('pair', market_data.get('pair', 'EURUSD'))
            
            signals[agent_name] = {
                'pair': pair,
                'signal': result.get('vote', 'HOLD'),
                'confidence': result.get('confidence', 50),
                'reasoning': result.get('reasoning', ''),
                'timestamp': datetime.now().isoformat()
            }
        
        return signals
    
    def _generate_consensus(self, signals: Dict) -> Dict:
        """
        Generate consensus from all signals
        """
        buy_votes = 0
        sell_votes = 0
        hold_votes = 0
        total_confidence = 0
        active_votes = 0
        reasoning = []
        confidence_by_signal = {'BUY': [], 'SELL': []}
        
        for agent_name, signal in signals.items():
            vote = signal.get('signal', 'HOLD')
            confidence = signal.get('confidence', 50)
            reason = signal.get('reasoning', '')
            
            if vote == 'BUY':
                buy_votes += 1
                total_confidence += confidence
                active_votes += 1
                confidence_by_signal['BUY'].append(confidence)
                if reason:
                    reasoning.append(f"{agent_name}: {reason[:50]}")
            elif vote == 'SELL':
                sell_votes += 1
                total_confidence += confidence
                active_votes += 1
                confidence_by_signal['SELL'].append(confidence)
                # if reason:
                #     reasoning.append(f"{agent_name}: {reason[:50]}")
            else:
                hold_votes += 1
        
        # Determine consensus with confidence weighting
        avg_buy_confidence = sum(confidence_by_signal['BUY']) / len(confidence_by_signal['BUY']) if confidence_by_signal['BUY'] else 0
        avg_sell_confidence = sum(confidence_by_signal['SELL']) / len(confidence_by_signal['SELL']) if confidence_by_signal['SELL'] else 0
        
        # Weighted score
        buy_score = buy_votes * (avg_buy_confidence / 100) if avg_buy_confidence > 0 else 0
        sell_score = sell_votes * (avg_sell_confidence / 100) if avg_sell_confidence > 0 else 0
        
        if buy_score > sell_score and buy_score > 0:
            signal = 'BUY'
            confidence = min(95, (buy_score / (buy_score + sell_score + 0.01)) * 100)
            reason = f"Consensus BUY ({buy_votes} votes, avg conf: {avg_buy_confidence:.0f}%)"
        elif sell_score > buy_score and sell_score > 0:
            signal = 'SELL'
            confidence = min(95, (sell_score / (buy_score + sell_score + 0.01)) * 100)
            reason = f"Consensus SELL ({sell_votes} votes, avg conf: {avg_sell_confidence:.0f}%)"
        else:
            signal = 'HOLD'
            confidence = 50
            reason = f"Consensus HOLD (BUY:{buy_votes} SELL:{sell_votes} HOLD:{hold_votes})"
        
        if reasoning:
            reason += f" - {'; '.join(reasoning[:2])}"
        
        return {
            'signal': signal,
            'confidence': round(confidence, 1),
            'reason': reason,
            'buy_votes': buy_votes,
            'sell_votes': sell_votes,
            'hold_votes': hold_votes,
            'buy_confidence': round(avg_buy_confidence, 1),
            'sell_confidence': round(avg_sell_confidence, 1),
            'total_agents': len(signals)
        }
    def _log_decision(self, results: Dict):
        """Log decision for analysis"""
        log_entry = {
            'timestamp': datetime.now().isoformat(),
            'cycle': results.get('cycle', 0),
            'consensus': results.get('consensus', {}),
            'trade': results.get('trade', {}),
            'engine': results.get('engine', {}),
            'agents': {
                name: {
                    'vote': r.get('vote', 'HOLD'),
                    'confidence': r.get('confidence', 50)
                }
                for name, r in results.get('agents', {}).items()
            }
        }
        
        self.decision_log.append(log_entry)
        
        # Keep log manageable
        if len(self.decision_log) > self.max_log_entries:
            self.decision_log.pop(0)
        
        # Save to file periodically
        if self.cycle_count % 10 == 0:
            try:
                with open('decision_log_v2.json', 'w') as f:
                    json.dump(self.decision_log[-100:], f, default=str, indent=2)
            except:
                pass
    def get_status(self) -> Dict:
        """
        Get current controller status
        """
        return {
            'name': self.name,
            'is_running': self.is_running,
            'cycle_count': self.cycle_count,
            'pairs': self.pairs,
            'agents': len(self.agents),
            # 'last_cycle': self.last_cycle_time.isoformat() if self.last_cycle_time else None,

            'anomaly_detector': self.anomaly_detector.get_protection_status() if hasattr(self.anomaly_detector, 'get_protection_status') else {},
            'execution': self.execution_engine.get_status() if hasattr(self.execution_engine, 'get_status') else {},
            'positions': self.execution_engine.positions if self.execution_engine else {},
            'trades_today': self.execution_engine.trades_today if self.execution_engine else 0,
            'daily_pnl': self.execution_engine.daily_pnl if self.execution_engine else 0,
            'timestamp': datetime.now().isoformat()
        }
    def get_agent_status(self) -> Dict:
        """Get status of all agents"""
        status = {}
        for agent in self.agents:
            if hasattr(agent, 'get_status'):
                try:
                    status[agent.name] = agent.get_status()
                except:
                    status[agent.name] = {'error': 'Could not get status'}
            else:
                status[agent.name] = {
                    'type': getattr(agent, 'agent_type', 'Unknown'),
                    'signal': getattr(agent, 'signal', 'HOLD'),
                    'confidence': getattr(agent, 'confidence', 50)
                }
        return status

    # ============================================================
    
    def run_backtest(self, historical_data: List[Dict]) -> List[Dict]:
        """Run backtest on historical data"""
        if not historical_data:
            logger.warning("No historical data provided")
            return []
        
        logger.info(f"Running backtest on {len(historical_data)} data points...")
        
        results = []
        for idx, data_point in enumerate(historical_data):
            try:
                result = self.process_cycle(data_point)
                results.append(result)
                
                if idx % 100 == 0:
                    logger.info(f"   Processed {idx}/{len(historical_data)}")
                    
            except Exception as e:
                logger.error(f"Error at {idx}: {e}")
        
        # Analyze results
        self._analyze_backtest(results)
        return results
    
    def _analyze_backtest(self, results: List[Dict]):
        """Analyze backtest results"""
        total_trades = 0
        wins = 0
        losses = 0
        
        for result in results:
            trade = result.get('trade', {})
            if trade.get('status') == 'EXECUTED':
                total_trades += 1
                if trade.get('pnl', 0) > 0:
                    wins += 1
                else:
                    losses += 1
        
        logger.info("=== BACKTEST RESULTS ===")
        logger.info(f"   Total trades: {total_trades}")
        logger.info(f"   Wins: {wins}")
        logger.info(f"   Losses: {losses}")
        if total_trades > 0:
            logger.info(f"   Win rate: {wins/total_trades*100:.1f}%")
        logger.info("==========================")
